  // منع التواريخ والأوقات القديمة
  function setupMinDateTime() {
    const now = new Date();
    const pad = n => String(n).padStart(2, '0');
    const localNow = `${now.getFullYear()}-${pad(now.getMonth()+1)}-${pad(now.getDate())}T${pad(now.getHours())}:${pad(now.getMinutes())}`;
    timeInput.min = localNow;

  function stateClass(status){
    if(status === 'reserved') return 'state-reserved';
    if(status === 'pending') return 'state-pending';
    if(status === 'selected') return 'state-selected';
    return 'state-available';
  }

  // مراجعة أوتوماتيكية لتحرير الطاولات (Auto-release) بعد 30 دقيقة
  function checkPendingExpirations() {
    const now = new Date().getTime();
    let changed = false;

    tables.forEach(t => {
      if (t.status === 'pending' && t.expireTime && now > t.expireTime) {
        t.status = 'available';
        t.expireTime = null;
        t.guestName = '';
        changed = true;
      }
    });

    if (changed) {
      renderMap();
    }
  }

  }

      btn.addEventListener('click', () => confirmCheckIn(t.id));
      pendingList.appendChild(btn);
    });
  }

  async function confirmCheckIn(tableId) {
    const table = tables.find(t => t.id === tableId);
    if (table) {
      table.status = 'reserved';
      table.expireTime = null;
      const guest = table.guestName;
      table.guestName = '';
      renderMap();

                           `👤 *Name:* ${guest}\n` +
                           `🪑 *Table:* ${table.id} is now officially *OCCUPIED* for the evening.`;
        
        await fetch(`https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage`, {

            text: checkInMsg,
            parse_mode: 'Markdown'
          })
        });
      }
    }
  }

  function handleTableClick(id){
    const table = tables.find(t => t.id === id);

    if(table.status === 'reserved' || table.status === 'pending'){
      const node = tablesLayer.querySelector(`[data-id="${id}"]`);
      node.classList.remove('shake');
      void node.offsetWidth;
      node.classList.add('shake');
      return;
    }

    if(selectedId === id){
      selectedId = null;
      resetForm();
      renderMap();
      return;
    }

    selectedId = id;
    renderMap();
    enableForm(table);
  }

  function enableForm(table){
    [nameInput, phoneInput, timeInput].forEach(i => i.disabled = false);
    setupMinDateTime();
    guestsInput.value = `${table.capacity} guests · Table ${table.id} (${table.shape})`;
    confirmBtn.disabled = false;
    selectionSummary.innerHTML = `Table <span class="text-[var(--gold)]">${table.id}</span> selected — ${table.zone}, seats ${table.capacity}.`;
  }

    bookingForm.reset();
    [nameInput, phoneInput, timeInput].forEach(i => i.disabled = true);
    guestsInput.value = '';
    confirmBtn.disabled = true;
    selectionSummary.textContent = 'No table selected yet.';
  }

        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          chat_id: TELEGRAM_CHAT_ID,
          text: message,
          parse_mode: 'Markdown'
        })
      });
    } catch(err) {
      console.warn('Telegram notification failed:', err);
    }
  }

  function formatDateTime(value){
    const d = new Date(value);
    if(isNaN(d)) return value;
    return d.toLocaleString(undefined, { weekday:'short', month:'short', day:'numeric', hour:'2-digit', minute:'2-digit' });
  }

  function showModal(details){
    document.getElementById('modalTable').textContent = details.table;
    document.getElementById('modalGuests').textContent = details.guests;
    document.getElementById('modalTime').textContent = details.time;
    document.getElementById('modalName').textContent = details.name;

    // صنع الـ QR الحقيقي والحي
    const qrContainer = document.getElementById('qrContainer');
    qrContainer.innerHTML = ''; 

    // نص الـ QR الحقيقي مع المعطيات ورابط تفاعلي
    const qrText = `RESERVATION_DETAILS\nName: ${details.name}\nPhone: ${details.phone}\nTable: ${details.tableId}\nTime: ${details.time}\nStatus: Pending`;
    
    // استعمال API ممتازة ومفتوحة لإنتاج الـ QR بالوقت وبدقة عالية
    const qrUrl = `https://api.qrserver.com/v1/create-qr-code/?size=150x150&data=${encodeURIComponent(qrText)}&color=141419&bgcolor=ffffff`;
    qrContainer.innerHTML = `<img src="${qrUrl}" alt="Booking QR" class="w-full h-full" />`;

    modalOverlay.classList.remove('hidden');
    modalOverlay.classList.add('flex');
  }
